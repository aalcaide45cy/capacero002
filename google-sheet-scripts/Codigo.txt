function alPegarEnlace(e) {
  try {
    if (!e || !e.range) return;
    var sheet = e.range.getSheet();
    var sheetName = sheet.getName();
    var row = e.range.getRow();
    var col = e.range.getColumn();
    if (row <= 1) return;

    var valorPegado = String(e.value || e.range.getValue()).trim();
    if (!valorPegado) return;

    // A. Si estamos en la pestaña "Modelos MakerWorld" o se pegó un enlace de MakerWorld
    if (sheetName === "Modelos MakerWorld" || valorPegado.indexOf("makerworld.com") !== -1) {
      if (col === 15 || sheetName === "Modelos MakerWorld") {
        procesarPegadoMakerWorld(sheet, row, valorPegado);
        return;
      }
    }

    // B. Si se pegó un enlace de YouTube
    if (sheetName !== "Modelos MakerWorld" && (valorPegado.indexOf("youtu.be") !== -1 || valorPegado.indexOf("youtube.com") !== -1)) {
      procesarPegadoYouTube(sheet, row, valorPegado);
      return;
    }
  } catch (err) {
    Logger.log("Error en alPegarEnlace: " + err.toString());
  }
}

/**
 * Traduce automáticamente cualquier texto (inglés u otro idioma) a español usando Google LanguageApp
 */
function traducirTextoAEspanol(texto) {
  if (!texto || typeof texto !== "string" || !texto.trim()) return "";
  try {
    var limpio = texto.trim();
    if (limpio.length <= 2500) {
      return LanguageApp.translate(limpio, "", "es");
    }
    var partes = limpio.split("\n\n");
    var traducidas = [];
    for (var p = 0; p < partes.length; p++) {
      var parrafo = partes[p].trim();
      if (parrafo) {
        try {
          traducidas.push(LanguageApp.translate(parrafo, "", "es"));
        } catch (eP) {
          traducidas.push(parrafo);
        }
      }
    }
    return traducidas.join("\n\n");
  } catch (err) {
    return texto;
  }
}

/**
 * Validador estricto de imágenes de modelos (descarta banderas public/us.png, avatares de usuario y platos)
 */
function esImagenValidaModelo(url) {
  if (!url || typeof url !== "string") return false;
  var u = url.toLowerCase().trim();
  if (u.indexOf("avatar") !== -1) return false;
  if (u.indexOf("/user/") !== -1) return false;
  if (u.indexOf("/public/") !== -1) return false;
  if (u.indexOf("/badge/") !== -1) return false;
  if (u.indexOf("plate_") !== -1) return false;
  if (u.indexOf(".svg") !== -1) return false;
  if (u.indexOf("us.png") !== -1 || u.indexOf("eu.png") !== -1) return false;
  return u.indexOf(".jpg") !== -1 || u.indexOf(".jpeg") !== -1 || u.indexOf(".png") !== -1 || u.indexOf(".webp") !== -1;
}

function extraerNombreArchivo(url) {
  if (!url || typeof url !== "string") return "";
  var clean = url.split("?")[0].split("#")[0];
  var parts = clean.split("/");
  return parts[parts.length - 1].toLowerCase();
}

function agregarImagenSinDuplicar(arrayImgs, vistosMap, url) {
  if (!url || !esImagenValidaModelo(url)) return;
  var fname = extraerNombreArchivo(url);
  if (!fname || vistosMap[fname]) return;
  vistosMap[fname] = true;
  arrayImgs.push(url);
}

/**
 * Extrae y estructura una fila completa de 19 columnas de MakerWorld (traducida y sin duplicados)
 */
function extraerFilaModeloMakerWorld(url, orderNum) {
  try {
    var isCrowdfunding = url.indexOf("/crowdfunding/") !== -1;
    var cfMatch = url.match(/\/crowdfunding\/(\d+)/i);
    var modelMatch = url.match(/\/models\/(\d+)/i) || url.match(/\/design\/(\d+)/i);

    var rawTitle = "";
    var rawDescription = "";
    var images = [];
    var vistosMap = {};
    var tag = "✨ NUEVO";
    var price = 0;
    var showPrice = 0;
    var buttonText = "IR A DISEÑO";
    var carouselInterval = 3;

    var requestHeaders = {
      "User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/128.0.0.0 Safari/537.36",
      "Accept": "application/json, text/plain, */*",
      "Accept-Language": "es-ES,es;q=0.9,en;q=0.8"
    };

    // CASO A: Proyecto Crowdfunding (/crowdfunding/230-...)
    if (isCrowdfunding && cfMatch) {
      var cfId = cfMatch[1];
      var cfApiUrl = "https://makerworld.com/api/v1/crowdfunding-service/project/" + cfId;
      var cfRes = UrlFetchApp.fetch(cfApiUrl, {
        headers: requestHeaders,
        muteHttpExceptions: true
      });

      if (cfRes.getResponseCode() === 200) {
        var cfData = JSON.parse(cfRes.getContentText());
        var basic = cfData.basic || {};
        var story = cfData.story || {};

        rawTitle = basic.title || "Proyecto Crowdfunding MakerWorld";
        
        var descSubtitle = basic.subtitle || "";
        var descStory = "";
        if (story.projectDescription) {
          descStory = String(story.projectDescription)
            .replace(/<br\s*[\/]?>/gi, "\n")
            .replace(/<\/p>/gi, "\n\n")
            .replace(/<[^>]+>/g, "")
            .replace(/&nbsp;/g, " ")
            .replace(/\n{3,}/g, "\n\n")
            .trim();
        }
        rawDescription = (descSubtitle ? descSubtitle + "\n\n" : "") + descStory;

        if (basic.image) agregarImagenSinDuplicar(images, vistosMap, basic.image);
        if (basic.bannerImage) agregarImagenSinDuplicar(images, vistosMap, basic.bannerImage);

        var jsonStrCf = JSON.stringify(cfData);
        var foundImgsCf = jsonStrCf.match(/https:\/\/[^"\\]+\.(?:jpg|jpeg|png|webp)/gi) || [];
        for (var cfI = 0; cfI < foundImgsCf.length; cfI++) {
          agregarImagenSinDuplicar(images, vistosMap, foundImgsCf[cfI]);
        }
      }
    } 
    // CASO B: Modelo 3D Estándar (/models/588782...)
    else if (modelMatch) {
      var modelId = modelMatch[1];
      var modelApiUrl = "https://makerworld.com/api/v1/design-service/design/" + modelId;
      var modelRes = UrlFetchApp.fetch(modelApiUrl, {
        headers: requestHeaders,
        muteHttpExceptions: true
      });

      if (modelRes.getResponseCode() === 200) {
        var modelData = JSON.parse(modelRes.getContentText());
        rawTitle = modelData.title || "Modelo MakerWorld";

        var rawDesc = modelData.summary || modelData.description || "";
        rawDescription = rawDesc
          .replace(/<br\s*[\/]?>/gi, "\n")
          .replace(/<\/p>/gi, "\n\n")
          .replace(/<[^>]+>/g, "")
          .replace(/&nbsp;/g, " ")
          .replace(/\n{3,}/g, "\n\n")
          .trim();

        if (modelData.coverUrl) agregarImagenSinDuplicar(images, vistosMap, modelData.coverUrl);

        if (modelData.designExtension && Array.isArray(modelData.designExtension.designImages)) {
          for (var d = 0; d < modelData.designExtension.designImages.length; d++) {
            agregarImagenSinDuplicar(images, vistosMap, modelData.designExtension.designImages[d].url);
          }
        }

        if (Array.isArray(modelData.instances)) {
          for (var inst = 0; inst < modelData.instances.length; inst++) {
            if (Array.isArray(modelData.instances[inst].pictures)) {
              for (var pic = 0; pic < modelData.instances[inst].pictures.length; pic++) {
                agregarImagenSinDuplicar(images, vistosMap, modelData.instances[inst].pictures[pic].url);
              }
            }
          }
        }

        var jsonStrModel = JSON.stringify(modelData);
        var foundImgsModel = jsonStrModel.match(/https:\/\/[^"\\]+\.(?:jpg|jpeg|png|webp)/gi) || [];
        for (var mI = 0; mI < foundImgsModel.length; mI++) {
          agregarImagenSinDuplicar(images, vistosMap, foundImgsModel[mI]);
        }
      }
    }

    if (!rawTitle) return null;

    // Traducción automática del título y la descripción al español
    var title = traducirTextoAEspanol(rawTitle);
    var description = traducirTextoAEspanol(rawDescription);

    return [
      orderNum,
      title,
      images[0] || "",
      images[1] || "",
      images[2] || "",
      images[3] || "",
      images[4] || "",
      images[5] || "",
      images[6] || "",
      images[7] || "",
      images[8] || "",
      images[9] || "",
      price,
      showPrice,
      url,
      description,
      tag,
      buttonText,
      carouselInterval
    ];
  } catch (err) {
    Logger.log("Error extraerFilaModeloMakerWorld: " + err.toString());
    return null;
  }
}

/**
 * ==============================================================================
 * 1. PROCESADOR OFICIAL DE MODELOS Y CROWDFUNDING DE MAKERWORLD (COLUMNAS A..S)
 * ==============================================================================
 */
function procesarPegadoMakerWorld(sheet, row, url) {
  try {
    // Calcular Order Correlativo Automático
    var nextOrder = 1;
    var lastRow = sheet.getLastRow();
    if (lastRow > 1) {
      var existingOrders = sheet.getRange(2, 1, lastRow - 1, 1).getValues();
      var maxOrder = 0;
      for (var o = 0; o < existingOrders.length; o++) {
        var val = parseInt(existingOrders[o][0], 10);
        if (!isNaN(val) && val > maxOrder && (o + 2) !== row) {
          maxOrder = val;
        }
      }
      nextOrder = maxOrder + 1;
    }

    var rowArray = extraerFilaModeloMakerWorld(url, nextOrder);
    if (rowArray) {
      sheet.getRange(row, 1, 1, 19).setValues([rowArray]);
    }
  } catch (err) {
    Logger.log("Error procesarPegadoMakerWorld: " + err.toString());
  }
}

function extraerIdUnicoMakerWorld(url) {
  if (!url || typeof url !== "string") return "";
  var cfMatch = url.match(/\/crowdfunding\/(\d+)/i);
  if (cfMatch) return "cf-" + cfMatch[1];
  var modelMatch = url.match(/\/models\/(\d+)/i) || url.match(/\/design\/(\d+)/i);
  if (modelMatch) return "model-" + modelMatch[1];
  return url.trim().toLowerCase();
}

/**
 * ==============================================================================
 * 5. MOTOR CRIPTOGRÁFICO VAPID ES256 (P-256 ECDSA) PARA GOOGLE APPS SCRIPT
 * ==============================================================================
 */
function bigintMod(a, m) {
  var r = a % m;
  return r >= ZERO ? r : r + m;
}

function bigintModInverse(k, m) {
  var m0 = m, y = ZERO, x = ONE;
  if (m === ONE) return ZERO;
  var a = bigintMod(k, m);
  while (a > ONE) {
    var q = a / m0, t = m0;
    m0 = a % m0; a = t;
    t = y;
    y = x - q * y;
    x = t;
  }
  if (x < ZERO) x += m;
  return x;
}

function ecPointAdd(p1, p2) {
  if (!p1) return p2;
  if (!p2) return p1;
  var x1 = p1[0], y1 = p1[1];
  var x2 = p2[0], y2 = p2[1];
  if (x1 === x2 && y1 === y2) return ecPointDouble(p1);
  if (x1 === x2) return null;
  var lambda = bigintMod((y2 - y1) * bigintModInverse(x2 - x1, P_CURVE), P_CURVE);
  var x3 = bigintMod(lambda * lambda - x1 - x2, P_CURVE);
  var y3 = bigintMod(lambda * (x1 - x3) - y1, P_CURVE);
  return [x3, y3];
}

function ecPointDouble(p) {
  if (!p) return null;
  var x = p[0], y = p[1];
  if (y === ZERO) return null;
  var lambda = bigintMod((THREE * x * x + A_CURVE) * bigintModInverse(TWO * y, P_CURVE), P_CURVE);
  var x3 = bigintMod(lambda * lambda - TWO * x, P_CURVE);
  var y3 = bigintMod(lambda * (x - x3) - y, P_CURVE);
  return [x3, y3];
}

function ecScalarMultiply(k, point) {
  var curr = point || [GX_CURVE, GY_CURVE];
  var result = null;
  var s = bigintMod(k, N_CURVE);
  while (s > ZERO) {
    if (s % TWO === ONE) result = ecPointAdd(result, curr);
    curr = ecPointDouble(curr);
    s = s / TWO;
  }
  return result;
}

function bytesToHex(bytes) {
  var hex = [];
  for (var i = 0; i < bytes.length; i++) {
    var b = bytes[i] < 0 ? bytes[i] + 256 : bytes[i];
    hex.push((b < 16 ? "0" : "") + b.toString(16));
  }
  return hex.join("");
}

function hexToBytes(hex) {
  var bytes = [];
  for (var c = 0; c < hex.length; c += 2) {
    bytes.push(parseInt(hex.substr(c, 2), 16));
  }
  return bytes;
}

function base64UrlEncodeBytes(bytes) {
  return Utilities.base64EncodeWebSafe(bytes).replace(/=+$/, "");
}

function base64UrlEncodeString(str) {
  return Utilities.base64EncodeWebSafe(str, Utilities.Charset.UTF_8).replace(/=+$/, "");
}

function base64UrlDecodeToHex(str) {
  var b64 = str.replace(/-/g, "+").replace(/_/g, "/");
  while (b64.length % 4) b64 += "=";
  var bytes = Utilities.base64Decode(b64);
  return bytesToHex(bytes);
}

function generarVapidJwt(aud, sub, privateKeyB64) {
  var header = { alg: "ES256", typ: "JWT" };
  var exp = Math.floor(Date.now() / 1000) + 86400;
  var payload = { aud: aud, exp: exp, sub: sub };
  
  var headerB64 = base64UrlEncodeString(JSON.stringify(header));
  var payloadB64 = base64UrlEncodeString(JSON.stringify(payload));
  var signingInput = headerB64 + "." + payloadB64;
  
  var msgHashBytes = Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_256, signingInput, Utilities.Charset.UTF_8);
  var msgHashHex = bytesToHex(msgHashBytes);
  var z = BigInt("0x" + msgHashHex);
  
  var privKeyHex = base64UrlDecodeToHex(privateKeyB64);
  var d = BigInt("0x" + privKeyHex);
  
  var r = ZERO, s = ZERO;
  for (var attempt = 0; attempt < 50; attempt++) {
    var seedBytes = Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_256, signingInput + attempt + Date.now(), Utilities.Charset.UTF_8);
    var k = bigintMod(BigInt("0x" + bytesToHex(seedBytes)), N_CURVE);
    if (k === ZERO) continue;
    
    var pt = ecScalarMultiply(k, [GX_CURVE, GY_CURVE]);
    if (!pt) continue;
    
    r = bigintMod(pt[0], N_CURVE);
    if (r === ZERO) continue;
    
    s = bigintMod(bigintModInverse(k, N_CURVE) * (z + r * d), N_CURVE);
    if (s !== ZERO) break;
  }
  
  if (s > N_CURVE / TWO) {
    s = N_CURVE - s;
  }
  
  var rHex = r.toString(16);
  while (rHex.length < 64) rHex = "0" + rHex;
  var sHex = s.toString(16);
  while (sHex.length < 64) sHex = "0" + sHex;
  
  var sigBytes = hexToBytes(rHex + sHex);
  var sigB64 = base64UrlEncodeBytes(sigBytes);
  
  return signingInput + "." + sigB64;
}

function extraerAudience(endpoint) {
  try {
    var match = endpoint.match(/^(https?:\/\/[^\/]+)/);
    return match ? match[1] : endpoint;
  } catch (e) {
    return endpoint;
  }
}

  var pcsEnviados = 0;
  var fallidos = 0;
  var audCache = {};
  
  for (var i = 0; i < data.length; i++) {
    var endpoint = String(data[i][2] || "").trim();
    var statusCol = numCols >= 14 ? 13 : 9;
    if (!endpoint || data[i][statusCol] === "CADUCADA") continue;
    
    var dispCol = numCols >= 14 ? 9 : 5;
    var osCol = numCols >= 14 ? 10 : 6;
    var disp = String(data[i][dispCol] || "").toLowerCase();
    var os = String(data[i][osCol] || "").toLowerCase();
    var isMobile = disp.indexOf("móvil") !== -1 || disp.indexOf("iphone") !== -1 || disp.indexOf("android") !== -1 || os.indexOf("ios") !== -1 || os.indexOf("android") !== -1;
    
    var aud = extraerAudience(endpoint);
    if (!audCache[aud]) {
      try {
        audCache[aud] = generarVapidJwt(aud, VAPID_SUBJECT, VAPID_PRIVATE_KEY);
      } catch (jwtErr) {
        audCache[aud] = null;
      }
    }
    
    var jwt = audCache[aud];
    
    try {
      var headers = {
        "TTL": "86400",
        "Urgency": "high"
      };
      
      if (jwt) {
        headers["Authorization"] = "vapid t=" + jwt + ", k=" + VAPID_PUBLIC_KEY;
        headers["Crypto-Key"] = "p256ecdsa=" + VAPID_PUBLIC_KEY;
      }
      
      var options = {
        method: "post",
        contentType: "application/json",
        headers: headers,
        payload: "",
        muteHttpExceptions: true
      };
      
      var response = UrlFetchApp.fetch(endpoint, options);
      var code = response.getResponseCode();
      
      if (code >= 400) {
        sheet.getRange(i + 2, statusCol + 1).setValue("CADUCADA");
        fallidos++;
      } else {
        enviados++;
        if (isMobile) movilesEnviados++; else pcsEnviados++;
      }
    } catch (err) {
      sheet.getRange(i + 2, statusCol + 1).setValue("CADUCADA");
      fallidos++;
    }
  }
  
  var histSheet = ss.getSheetByName("Historial_Notificaciones");
  if (!histSheet) {
    histSheet = ss.insertSheet("Historial_Notificaciones");
    crearYFormatearPestanaHistorial(false);
  }
  
  var histRow = [
    new Date().toISOString().replace('T', ' ').substring(0, 19),
    title,
    body,
    url || "https://www.capacero3d.com",
    movilesEnviados,
    pcsEnviados,
    enviados,
    fallidos
  ];
  histSheet.appendRow(histRow);
  
  return {
    success: true,
    message: "✅ ¡Notificación enviada a " + enviados + " dispositivo(s)! (" + movilesEnviados + " móviles, " + pcsEnviados + " PCs" + (fallidos > 0 ? " y " + fallidos + " caducados marcados" : "") + ")"
  };
}

/**
 * ==============================================================================
 * 9. DETECTOR ESTRICTO ANTI-SHORTS
 * ==============================================================================
 */
function esShort(videoId, url, title, durationSec) {
  var urlLower = (url || "").toLowerCase();
  var titleLower = (title || "").toLowerCase();
  
  if (VIDEOS_IGNORADOS.indexOf(videoId) !== -1) return true;
  if (urlLower.indexOf("/shorts/") !== -1) return true;
  if (titleLower.indexOf("#shorts") !== -1 || titleLower.indexOf("#short") !== -1) return true;
  
  if (durationSec && durationSec > 0) {
    return durationSec <= 60;
  }
  
  return false;
}

function extraerYouTubeId(url) {
  var regExp = /^.*(youtu\.be\/|v\/|u\/\w\/|embed\/|watch\?v=|&v=)([^#&?]*).*/;
  var match = url.match(regExp);
  return (match && match[2].length === 11) ? match[2] : null;
}

      // B. Descripción completa con regex de shortDescription (método de la fila 29)
      var matchDesc = html.match(/"shortDescription":"(.*?)(?<!\\)"/);
      if (matchDesc && matchDesc[1]) {
        try {
          description = JSON.parse('"' + matchDesc[1] + '"');
        } catch (e) {
          description = matchDesc[1]
            .replace(/\\n/g, '\n')
            .replace(/\\r/g, '')
            .replace(/\\"/g, '"')
            .replace(/\\\\/g, '\\');
        }
      }

      // C. Extracción por analizador de llaves sobre ytInitialPlayerResponse (inmune a saltos de línea)
      var idxPlayer = html.indexOf("ytInitialPlayerResponse = ");
      if (idxPlayer !== -1) {
        var start = idxPlayer + "ytInitialPlayerResponse = ".length;
        var count = 0;
        var end = -1;
        for (var p = start; p < start + 350000 && p < html.length; p++) {
          if (html.charAt(p) === '{') count++;
          else if (html.charAt(p) === '}') {
            count--;
            if (count === 0) {
              end = p + 1;
              break;
            }
          }
        }
        if (end !== -1) {
          try {
            var pObj = JSON.parse(html.slice(start, end));
            if (pObj && pObj.videoDetails) {
              if (!title && pObj.videoDetails.title) title = pObj.videoDetails.title;
              if ((!description || description.length < 50) && pObj.videoDetails.shortDescription) {
                description = pObj.videoDetails.shortDescription;
              }
              if (pObj.videoDetails.keywords) keywords = pObj.videoDetails.keywords;
              if (pObj.videoDetails.lengthSeconds) durationSec = parseInt(pObj.videoDetails.lengthSeconds, 10);
            }
          } catch (eP) {}
        }
      }

      // D. Duración para comprobación de Shorts
      if (!durationSec) {
        var matchDur = html.match(/"approxDurationMs":"(\d+)"/);
        if (matchDur && matchDur[1]) {
          durationSec = Math.round(parseInt(matchDur[1], 10) / 1000);
        }
      }
    }
  } catch (eHtml) {
    Logger.log("Aviso Scraping HTML: " + eHtml.toString());
  }

  // MÉTODO 2: API Innertube con headers completos de navegador (si aún faltase descripción o título)
  if (!description || !title) {
    try {
      var payload = JSON.stringify({
        videoId: videoId,
        context: {
          client: {
            clientName: "WEB",
            clientVersion: "2.20240101.00.00",
            hl: "es",
            gl: "ES"
          }
        }
      });

      var resIt = UrlFetchApp.fetch("https://www.youtube.com/youtubei/v1/player?prettyPrint=false", {
        method: "post",
        contentType: "application/json",
        payload: payload,
        muteHttpExceptions: true,
        headers: {
          "User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/128.0.0.0 Safari/537.36",
          "X-YouTube-Client-Name": "1",
          "X-YouTube-Client-Version": "2.20240101.00.00",
          "Origin": "https://www.youtube.com",
          "Referer": "https://www.youtube.com/watch?v=" + videoId
        }
      });

      if (resIt.getResponseCode() === 200) {
        var dataIt = JSON.parse(resIt.getContentText());
        var details = dataIt.videoDetails;
        if (details) {
          if (!title && details.title) title = details.title;
          if (!description && details.shortDescription) description = details.shortDescription;
          if (!durationSec && details.lengthSeconds) durationSec = parseInt(details.lengthSeconds, 10);
          if ((!keywords || keywords.length === 0) && Array.isArray(details.keywords)) keywords = details.keywords;
        }
      }
    } catch (eIt) {
      Logger.log("Aviso Innertube: " + eIt.toString());
    }
  }

  // MÉTODO 3: YouTube oEmbed Oficial (garantiza el título si no se obtuvo)
  if (!title) {
    try {
      var oembedUrl = "https://www.youtube.com/oembed?url=https://www.youtube.com/watch?v=" + videoId + "&format=json";
      var resOe = UrlFetchApp.fetch(oembedUrl, { muteHttpExceptions: true });
      if (resOe.getResponseCode() === 200) {
        var dataOe = JSON.parse(resOe.getContentText());
        if (dataOe.title) title = dataOe.title;
      }
    } catch (eOe) {}
  }

  // MÉTODO 4: noembed.com fallback para el título
  if (!title) {
    try {
      var apiUrl = "https://noembed.com/embed?url=https://www.youtube.com/watch?v=" + videoId;
      var resApi = UrlFetchApp.fetch(apiUrl, { muteHttpExceptions: true });
      if (resApi.getResponseCode() === 200) {
        var dataApi = JSON.parse(resApi.getContentText());
        if (dataApi.title) title = dataApi.title;
      }
    } catch (eNo) {}
  }

  if (durationSec > 0 && durationSec <= 60) {
    isShort = true;
  }

  if (!title) {
    title = "Tutorial #" + videoId;
  }

  // MÉTODO 5: Respaldo de seguridad con el catálogo del canal Capa Cero
  if (!description || description.length < 50 || description.indexOf("Tutorial oficial de Capa Cero 3D (") === 0) {
    var catalogo = obtenerCatalogoOficialCapaCero();
    for (var c = 0; c < catalogo.length; c++) {
      if (catalogo[c][1].indexOf(videoId) !== -1) {
        if (!title || title.indexOf("Tutorial #") === 0) title = catalogo[c][0];
        category = catalogo[c][2];
        description = catalogo[c][3];
        break;
      }
    }
  }

  if (!description) {
    description = "Tutorial oficial de Capa Cero 3D (" + title + "). Guía paso a paso para dominar tu impresora 3D y el laminador.";
  }

  // Detectar categoría precisa combinando título, keywords y descripción
  var textoCompleto = (title + " " + keywords.join(" ") + " " + description).toLowerCase();
  if (textoCompleto.indexOf("fusion") !== -1 || textoCompleto.indexOf("360") !== -1 || textoCompleto.indexOf("cad") !== -1 || textoCompleto.indexOf("autodesk") !== -1) {
    category = "Modelado 3D";
  } else if (textoCompleto.indexOf("filamento") !== -1 || textoCompleto.indexOf("perfil") !== -1 || textoCompleto.indexOf("costura") !== -1 || textoCompleto.indexOf("calibrac") !== -1 || textoCompleto.indexOf("flujo") !== -1) {
    category = "Perfiles y Calibración";
  } else if (textoCompleto.indexOf("multicolor") !== -1 || textoCompleto.indexOf("ams") !== -1 || textoCompleto.indexOf("pintar") !== -1 || textoCompleto.indexOf("purga") !== -1) {
    category = "Multicolor y AMS";
  } else if (textoCompleto.indexOf("boquilla") !== -1 || textoCompleto.indexOf("hardware") !== -1 || textoCompleto.indexOf("laser") !== -1 || textoCompleto.indexOf("grabador") !== -1 || textoCompleto.indexOf("hotend") !== -1) {
    category = textoCompleto.indexOf("laser") !== -1 ? "Grabado Láser" : "Hardware y Boquillas";
  } else if (textoCompleto.indexOf("ahorra") !== -1 || textoCompleto.indexOf("tiempo") !== -1 || textoCompleto.indexOf("truco") !== -1 || textoCompleto.indexOf("chatgpt") !== -1 || textoCompleto.indexOf("dinero") !== -1 || textoCompleto.indexOf("secreto") !== -1) {

  } else {
    category = "Bambu Studio";
  }

  var descargas = extraerDescargasDeDescripcion(description);
  var consejoClave = generarConsejoClaveContextual(title, description, category);

  return {
    videoId: videoId,
    title: title,
    category: category,
    description: description,
    consejoClave: consejoClave,
    durationSec: durationSec,
    isShort: isShort,
    descargas: descargas
  };
}

/**
 * Extrae enlaces de descarga, plantillas o modelos de la descripción
 */
function extraerDescargasDeDescripcion(desc) {
  if (!desc) return [];
  var enlaces = [];
  var vistos = {};

  // 1. Buscar enlaces a MakerWorld
  var mwMatches = desc.match(/https:\/\/(?:www\.)?makerworld\.com\/[^\s\)\"\'<>]+/gi) || [];
  for (var i = 0; i < mwMatches.length; i++) {
    var u = mwMatches[i].replace(/[.,;:]+$/, "");
    if (!vistos[u]) {
      vistos[u] = true;
      enlaces.push(u);
    }
  }

  // 2. Buscar enlaces de recursos, plantillas o descargas (capacero3d, drive, etc.)
  var lines = desc.split("\n");
  for (var j = 0; j < lines.length; j++) {
    var l = lines[j];
    if (/descarga|plantilla|modelo|recurso|archivo|gu[ií]a/i.test(l)) {
      var urlMatches = l.match(/https?:\/\/[^\s\)\"\'<>]+/gi) || [];
      for (var k = 0; k < urlMatches.length; k++) {
        var clean = urlMatches[k].replace(/[.,;:]+$/, "");
        if (!clean.includes("youtube.com") && !clean.includes("youtu.be") && !vistos[clean]) {
          vistos[clean] = true;
          enlaces.push(clean);
        }
      }
    }
  }

  return enlaces;
}

function sincronizarVideosCapaCero() {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var catalogoCanal = obtenerCatalogoOficialCapaCero();
  
  var existingUrls = {};
  var lastRow = sheet.getLastRow();
  if (lastRow > 1) {
    var data = sheet.getRange(2, 2, lastRow - 1, 1).getValues();
    for (var i = 0; i < data.length; i++) {
      if (data[i][0]) {
        existingUrls[String(data[i][0]).trim()] = true;
      }
    }
  }
  
  var totalNuevos = 0;
  for (var k = 0; k < catalogoCanal.length; k++) {
    var videoUrl = catalogoCanal[k][1];
    if (!existingUrls[videoUrl]) {
      sheet.appendRow(catalogoCanal[k]);
      existingUrls[videoUrl] = true;
      totalNuevos++;
    }
  }
  
  SpreadsheetApp.getUi().alert(
    totalNuevos > 0 
      ? "✅ ¡Sincronizados " + totalNuevos + " tutoriales horizontales en español!" 
      : "ℹ️ Tu hoja ya contiene exactamente los 29 tutoriales horizontales."
  );
}

function limpiarYResincronizar() {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var lastRow = sheet.getLastRow();
  if (lastRow > 1) {
    sheet.getRange(2, 1, lastRow - 1, 9).clearContent();
  }
  sincronizarVideosCapaCero();
}

function obtenerOcrearPestanaBovedas(ss) {
  if (!ss) ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName("Bovedas_Sync");
  if (!sheet) {
    sheet = ss.insertSheet("Bovedas_Sync");
    sheet.appendRow(["VaultId", "Ultima_Actualizacion", "Dispositivo", "Payload_JSON"]);
    sheet.getRange(1, 1, 1, 4).setFontWeight("bold").setBackground("#0f172a").setFontColor("#38bdf8");
    sheet.setFrozenRows(1);
    sheet.setColumnWidth(1, 160);
    sheet.setColumnWidth(2, 180);
    sheet.setColumnWidth(3, 130);
    sheet.setColumnWidth(4, 400);
  }
  return sheet;
}
